import { NextResponse } from "next/server";
import { revalidatePath } from "next/cache";
import Project from "@/models/Project";
import { getProjectFileBucket } from "@/lib/gridfs";
import {
  sanitizeFilename,
  validateProjectFiles,
} from "@/lib/project-files";
import { requireCurrentUser } from "@/lib/current-user";

export const runtime = "nodejs";

const uploadFile = async ({ bucket, file, creatorId, projectId }) => {
  const filename = sanitizeFilename(file.name);
  const contentType = file.type || "application/octet-stream";
  const bytes = Buffer.from(await file.arrayBuffer());

  return new Promise((resolve, reject) => {
    const upload = bucket.openUploadStream(filename, {
      contentType,
      metadata: {
        creatorId: String(creatorId),
        projectId: String(projectId),
      },
    });

    upload.once("error", reject);
    upload.once("finish", () => {
      resolve({
        fileId: upload.id,
        filename,
        contentType,
        size: file.size,
      });
    });
    upload.end(bytes);
  });
};

export async function POST(request) {
  const uploadedIds = [];
  let project;

  try {
    const user = await requireCurrentUser();
    const formData = await request.formData();
    const title = String(formData.get("title") || "").trim();
    const description = String(formData.get("description") || "").trim();
    const files = formData
      .getAll("files")
      .filter((file) => typeof file?.arrayBuffer === "function");

    if (title.length < 3 || title.length > 80) {
      return NextResponse.json(
        { ok: false, error: "Project title must contain between 3 and 80 characters." },
        { status: 400 },
      );
    }

    if (description.length > 500) {
      return NextResponse.json(
        { ok: false, error: "Project description must contain at most 500 characters." },
        { status: 400 },
      );
    }

    const fileError = validateProjectFiles(files);

    if (fileError) {
      return NextResponse.json(
        { ok: false, error: fileError },
        { status: 400 },
      );
    }

    const bucket = await getProjectFileBucket();
    project = new Project({
      creatorId: user._id,
      title,
      description,
      files: [],
    });

    const uploadedFiles = [];

    for (const file of files) {
      const uploadedFile = await uploadFile({
        bucket,
        file,
        creatorId: user._id,
        projectId: project._id,
      });

      uploadedIds.push(uploadedFile.fileId);
      uploadedFiles.push(uploadedFile);
    }

    project.files = uploadedFiles;
    await project.save();
    revalidatePath(`/${user.username}`);
    revalidatePath("/dashboard");

    return NextResponse.json({
      ok: true,
      message: "Project folder uploaded successfully.",
      projectId: String(project._id),
    });
  } catch (error) {
    try {
      const bucket = await getProjectFileBucket();
      await Promise.allSettled(uploadedIds.map((fileId) => bucket.delete(fileId)));
    } catch {
      // The original upload error is more useful than cleanup failures.
    }

    return NextResponse.json(
      { ok: false, error: error.message || "Could not upload the project folder." },
      { status: error.message?.includes("signed in") ? 401 : 500 },
    );
  }
}

import mongoose from "mongoose";
import { NextResponse } from "next/server";
import { revalidatePath } from "next/cache";
import Project from "@/models/Project";
import { getProjectFileBucket } from "@/lib/gridfs";
import { requireCurrentUser } from "@/lib/current-user";

export const runtime = "nodejs";

export async function DELETE(_request, { params }) {
  try {
    const { projectId } = await params;

    if (!mongoose.isValidObjectId(projectId)) {
      return NextResponse.json(
        { ok: false, error: "Invalid project folder identifier." },
        { status: 400 },
      );
    }

    const user = await requireCurrentUser();
    const project = await Project.findOne({
      _id: projectId,
      creatorId: user._id,
    });

    if (!project) {
      return NextResponse.json(
        { ok: false, error: "Project folder not found." },
        { status: 404 },
      );
    }

    const bucket = await getProjectFileBucket();
    await Promise.allSettled(
      project.files.map((file) => bucket.delete(file.fileId)),
    );
    await Project.deleteOne({ _id: project._id });
    revalidatePath(`/${user.username}`);
    revalidatePath("/dashboard");

    return NextResponse.json({
      ok: true,
      message: "Project folder deleted.",
    });
  } catch (error) {
    return NextResponse.json(
      { ok: false, error: error.message || "Could not delete the project folder." },
      { status: error.message?.includes("signed in") ? 401 : 500 },
    );
  }
}

import { Readable } from "node:stream";
import mongoose from "mongoose";
import Project from "@/models/Project";
import connectDb from "@/db/connectDB";
import { getProjectFileBucket } from "@/lib/gridfs";

export const runtime = "nodejs";

const INLINE_TYPES = new Set([
  "application/pdf",
  "image/gif",
  "image/jpeg",
  "image/png",
  "image/webp",
  "text/markdown",
  "text/plain",
]);

const contentDisposition = (filename, inline) => {
  const safeAscii = filename.replace(/[^a-zA-Z0-9._ -]/g, "_");
  const encoded = encodeURIComponent(filename);

  return `${inline ? "inline" : "attachment"}; filename="${safeAscii}"; filename*=UTF-8''${encoded}`;
};

export async function GET(_request, { params }) {
  const { fileId } = await params;

  if (!mongoose.isValidObjectId(fileId)) {
    return new Response("Invalid file identifier.", { status: 400 });
  }

  await connectDb();

  const objectId = new mongoose.Types.ObjectId(fileId);
  const project = await Project.findOne({ "files.fileId": objectId })
    .select("files")
    .lean();

  if (!project) {
    return new Response("Project file not found.", { status: 404 });
  }

  const file = project.files.find(
    (candidate) => String(candidate.fileId) === fileId,
  );

  if (!file) {
    return new Response("Project file not found.", { status: 404 });
  }

  const bucket = await getProjectFileBucket();
  const nodeStream = bucket.openDownloadStream(objectId);
  const webStream = Readable.toWeb(nodeStream);

  return new Response(webStream, {
    headers: {
      "Cache-Control": "public, max-age=3600",
      "Content-Disposition": contentDisposition(
        file.filename,
        INLINE_TYPES.has(file.contentType),
      ),
      "Content-Length": String(file.size),
      "Content-Type": file.contentType,
      "X-Content-Type-Options": "nosniff",
    },
  });
}

import { randomUUID } from "node:crypto";
import { NextResponse } from "next/server";
import connectDb from "@/db/connectDB";
import Payment from "@/models/Payment";
import User from "@/models/User";
import { validateContributionInput } from "@/lib/contribution-validation";
import {

} from "@/lib/razorpay-test";

export const runtime = "nodejs";

export async function POST(request) {
  try {
    const input = await request.json();
    const validation = validateContributionInput(input);

    if (!validation.ok) {
      return NextResponse.json(validation, { status: 400 });
    }

    const { username, name, message, amount } = validation.value;
    const { keyId } = getRazorpayTestCredentials();

    await connectDb();

    const creator = await User.findOne({ username })
      .select("_id name username")
      .lean();

    if (!creator) {
      return NextResponse.json(
        { ok: false, error: "This creator page no longer exists." },
        { status: 404 },
      );
    }

    const receipt = `gf_${Date.now()}_${randomUUID().slice(0, 8)}`;
    const order = await razorpayRequest("/orders", {
      method: "POST",
      body: JSON.stringify({
        amount: amount * 100,
        currency: "INR",
        receipt,
        notes: {
          creator: creator.username,
          mode: "GetFunds portfolio test",
        },
      }),
    });

    if (!order?.id || order.amount !== amount * 100) {
      throw new Error("Razorpay returned an invalid test order.");
    }

    await Payment.create({
      creatorId: creator._id,
      name,
      to_user: creator.username,
      oid: order.id,
      message,
      amount,
      mode: "razorpay-test",
      status: "pending",
      done: false,
      expiresAt: new Date(Date.now() + 24 * 60 * 60 * 1000),
    });

    return NextResponse.json({
      ok: true,
      checkout: {
        keyId,
        orderId: order.id,
        amount: order.amount,
        currency: order.currency,
        creatorName: creator.name || creator.username,
        supporterName: name,
      },
    });
  } catch (error) {
    const configurationError =
      error.message?.includes("not configured") ||
      error.message?.includes("Only Razorpay Test Mode");

    return NextResponse.json(
      { ok: false, error: error.message || "Could not create a Razorpay test order." },
      { status: configurationError ? 503 : 500 },
    );
  }
}

import {
  createHmac,
  timingSafeEqual,
} from "node:crypto";
import { revalidatePath } from "next/cache";
import { NextResponse } from "next/server";
import connectDb from "@/db/connectDB";
import Payment from "@/models/Payment";
import {
  getRazorpayTestCredentials,
  razorpayRequest,
} from "@/lib/razorpay-test";

export const runtime = "nodejs";

const safeSignatureMatch = (expected, received) => {
  if (!/^[a-f0-9]{64}$/i.test(received)) {
    return false;
  }

  const expectedBuffer = Buffer.from(expected, "hex");
  const receivedBuffer = Buffer.from(received, "hex");

  return (
    expectedBuffer.length === receivedBuffer.length &&
    timingSafeEqual(expectedBuffer, receivedBuffer)
  );
};

export async function POST(request) {
  try {
    const body = await request.json();
    const orderId = String(body?.razorpay_order_id || "");
    const paymentId = String(body?.razorpay_payment_id || "");
    const signature = String(body?.razorpay_signature || "");

    if (!/^order_[a-zA-Z0-9]+$/.test(orderId)) {
      return NextResponse.json(
        { ok: false, error: "Invalid Razorpay test order identifier." },
        { status: 400 },
      );
    }

    if (!/^pay_[a-zA-Z0-9]+$/.test(paymentId)) {
      return NextResponse.json(
        { ok: false, error: "Invalid Razorpay test payment identifier." },
        { status: 400 },
      );
    }

    const { keySecret } = getRazorpayTestCredentials();
    await connectDb();

    const pendingPayment = await Payment.findOne({
      oid: orderId,
      mode: "razorpay-test",
    });

    if (!pendingPayment) {
      return NextResponse.json(
        { ok: false, error: "The pending test contribution was not found." },
        { status: 404 },
      );
    }

    if (pendingPayment.done) {
      if (pendingPayment.paymentId === paymentId) {
        return NextResponse.json({
          ok: true,
          message: "This Razorpay test payment was already verified.",
        });
      }

      return NextResponse.json(
        { ok: false, error: "This test order has already been completed." },
        { status: 409 },
      );
    }

    const expectedSignature = createHmac("sha256", keySecret)
      .update(`${pendingPayment.oid}|${paymentId}`)
      .digest("hex");

    if (!safeSignatureMatch(expectedSignature, signature)) {
      return NextResponse.json(
        { ok: false, error: "Razorpay payment signature verification failed." },
        { status: 400 },
      );
    }

    const remotePayment = await razorpayRequest(`/payments/${paymentId}`);
    const acceptedStatuses = new Set(["authorized", "captured"]);

    if (
      remotePayment?.order_id !== pendingPayment.oid ||
      remotePayment?.amount !== pendingPayment.amount * 100 ||
      remotePayment?.currency !== "INR" ||
      !acceptedStatuses.has(remotePayment?.status)
    ) {
      return NextResponse.json(
        { ok: false, error: "Razorpay could not confirm the expected test payment." },
        { status: 400 },
      );
    }

    const completedPayment = await Payment.findOneAndUpdate(
      { _id: pendingPayment._id, done: false },
      {
        $set: {
          paymentId,
          status: "completed",
          done: true,
        },
        $unset: {
          expiresAt: 1,
        },
      },
      { new: true },
    );

    if (!completedPayment) {
      return NextResponse.json(
        { ok: false, error: "This test payment was processed concurrently." },
        { status: 409 },
      );
    }

    revalidatePath(`/${pendingPayment.to_user}`);
    revalidatePath("/dashboard");

    return NextResponse.json({
      ok: true,
      message: "Razorpay test payment verified. No real money was moved.",
    });
  } catch (error) {
    const configurationError =
      error.message?.includes("not configured") ||
      error.message?.includes("Only Razorpay Test Mode");

    return NextResponse.json(
      { ok: false, error: error.message || "Could not verify the Razorpay test payment." },
      { status: configurationError ? 503 : 500 },
    );
  }
}

# GetFunds resume and interview guide

> **Historical interview guide:** this document maps the original Razorpay prototype. Use [05-current-demo-architecture.md](./05-current-demo-architecture.md) for defensible claims about the current implementation.

## One-sentence description

GetFunds is a full-stack Next.js crowdfunding application where creators authenticate with GitHub, configure a username-based fundraising page, and accept INR contributions through Razorpay while MongoDB stores creator profiles and payment state.

## 60-second interview pitch

> I built GetFunds as a full-stack crowdfunding platform using the Next.js App Router. A creator signs in through GitHub OAuth using NextAuth, and on first login the application provisions a local MongoDB user. The creator configures a public username-based fundraising page and their Razorpay account. A supporter can visit that page without logging in, submit a contribution, and the server creates a Razorpay order while recording a pending payment in MongoDB. The public page queries completed payments to calculate the raised amount and show the top ten supporters. I used server components for route validation and dynamic metadata, client components for interactive forms and checkout, and server actions for database and Razorpay operations. The project also taught me that payment verification, idempotency, strict server-side authorization, and secret handling are essential before production deployment.

That pitch describes the actual code and proactively shows good engineering judgment about its current limits.

## System flow to draw in an interview

```mermaid
flowchart LR
    Creator[Creator] -->|GitHub OAuth| Auth[NextAuth]
    Auth -->|provision and session| User[(MongoDB User)]
    Creator -->|configure page| Dashboard[Creator dashboard]
    Dashboard --> User

    Supporter[Supporter] --> Page[Dynamic username page]
    Page -->|read completed contributions| Payments[(MongoDB Payments)]
    Supporter -->|submit contribution| ServerAction[Next.js server action]
    ServerAction -->|create order| Razorpay[Razorpay]
    ServerAction -->|store pending order| Payments
    Razorpay -.->|verified callback or webhook needed| Payments
```

When drawing it, explain the solid path first. Then mention that the dashed arrow is the critical completion path and that the current repository still needs the callback handler.

## Resume claims mapped to evidence

### Header

> GetFunds | Next.js, MongoDB, Razorpay, NextAuth.js

This technology list is supported by `package.json` and the implementation:

- Next.js 15 App Router provides routes, server components, route handlers, and server actions.
- MongoDB is accessed through Mongoose and stores `User` and `Payment` documents.
- Razorpay's server SDK creates INR orders, and its checkout script opens the hosted payment UI.
- NextAuth.js handles GitHub OAuth and sessions.

### Resume bullet 1

> Developed a full-stack crowdfunding platform using Next.js and MongoDB, enabling developers to create and manage concurrent fundraising campaigns.

#### What the code actually does

- The frontend, backend operations, authentication endpoint, and database layer are in one Next.js application.
- `User` stores the creator profile and payment configuration.
- `Payment` stores contributions for a creator using `to_user`.
- The dynamic route `/{username}` provides one public fundraising page for every creator.
- `Dashboard` lets a creator manage page/profile settings.
- Many creators can have active pages and receive independent payment orders at the same time.

#### What “concurrent campaigns” means here

The current version treats a creator's public page as the campaign. It supports many creators' pages concurrently, but it does not have a `Campaign` model and cannot create multiple separately named campaigns for one creator.

If questioned, say:

> In the first version, I modeled a campaign implicitly as a creator's username page. That allowed multiple creators to fundraise concurrently with a simple one-to-many User-to-Payment design. For multiple campaigns per creator, I would add a Campaign collection and make payments reference an immutable campaign ID.

#### Stronger, precise resume wording

> Built a full-stack crowdfunding platform with Next.js and MongoDB, enabling multiple creators to configure and operate independent username-based fundraising pages.

Use the original sentence only if you later implement an explicit `Campaign` entity.

### Resume bullet 2

> Integrated Razorpay for secure financial transactions and built a dynamic dashboard for real-time funding tracking.

#### Razorpay flow implemented in the code

1. The browser converts rupees to paise and calls `initiate`.
2. The server loads the target creator's Razorpay credentials.
3. The Razorpay server SDK creates an INR order.
4. MongoDB receives a `Payment` record with the order ID and `done: false`.
5. The server returns the order to the browser.
6. The Razorpay checkout script opens using the public key ID and order ID.
7. `fetchpayments` displays only records where `done: true`.

This pending-then-verified design is the correct shape: unverified orders should never count toward funds raised.

#### What is not yet implemented

`PaymentPage` points Razorpay to `/api/razorpay`, but that route does not exist in the repository. There is therefore no code that verifies the Razorpay signature and changes `done` to `true`. The browser query parameter `paymentdone=true` only shows a toast and must never be treated as payment proof.

Also, the current dashboard is a profile/settings dashboard. Funding totals are calculated on the public creator page when its client component loads. The data is dynamic, but it is not live real-time data: there are no WebSockets, Server-Sent Events, change streams, or polling.

#### How to explain secure payment processing

Say that secure processing requires all of the following:

- create orders server-side so the secret is not in browser code;
- use the provider-hosted checkout so card details do not pass through GetFunds;
- store an internal pending record linked by provider order ID;
- verify the callback/webhook HMAC signature on the server;
- compare expected amount, currency, and order identity;
- make completion idempotent so retries do not double-count a payment; and
- expose only completed payments in funding totals.

The project currently implements order creation, hosted checkout, pending persistence, and completed-only reads. Be clear that callback verification is the required next piece.

#### Stronger, precise resume wording for the current code

> Integrated Razorpay order creation and hosted checkout, persisted pending payment records in MongoDB, and built a dynamic creator page that aggregates completed funding and ranks the top ten supporters.

After implementing callback verification and a live dashboard, you can safely restore the stronger “secure transactions” and “real-time tracking” language.

### Resume bullet 3

> Implemented NextAuth.js for secure OAuth login and leveraged Server-Side Rendering (SSR) to optimize SEO and performance.

#### OAuth flow implemented in the code

1. `signIn("github")` starts authentication from the login page.
2. NextAuth redirects the visitor to GitHub.
3. GitHub returns identity/account data to the catch-all NextAuth route.
4. The sign-in callback connects to MongoDB and tries to find or create the local user.
5. The session callback loads the local username and exposes it as `session.user.name`.
6. `SessionProvider` makes session state available to client components.
7. The navbar changes its actions, and the dashboard redirects unauthenticated visitors.

OAuth is preferable to storing passwords in this project because GitHub handles credential verification and GetFunds receives an identity after authorization. NextAuth also handles state/CSRF protections and session mechanics.

#### SSR behavior implemented in the code

- `app/[username]/page.js` is a server component.
- It queries MongoDB before returning and calls `notFound()` for an invalid username.
- `generateMetadata` builds a server-side title such as `Support alice - Get Me A Chai`.
- The landing and about pages are server components by default.

However, the visible creator record, payments, total, and leaderboard are fetched in `PaymentPage` after the client mounts. This means the most important creator content is client-rendered, not fully SSR.

#### Why server rendering helps

- Search engines and link previews can receive route-specific metadata.
- The server can reject an invalid creator route before sending a successful page.
- Server-rendered content can reduce the time to meaningful first paint and the amount of client JavaScript when data fetching stays in server components.
- Server-only code can access the database without exposing database credentials.

#### Stronger, precise resume wording

> Implemented GitHub OAuth and session-aware navigation with NextAuth.js, and used Next.js server components plus dynamic metadata for server-side route validation and page-specific SEO.

If you move `fetchuser` and `fetchpayments` into the server route and pass safe data into `PaymentPage`, then the broader SSR claim will be much more defensible.

## Deep technical explanation by feature

### Why use server actions?

Server actions let client components call server-side operations without creating a separate REST endpoint for every form. They are useful here for profile updates, database reads, and Razorpay order creation. They are still network-accessible server entry points, so the application must validate input, verify the session for protected actions, and authorize the affected record.

### Why maintain a local user after OAuth?

The OAuth provider owns identity data, while GetFunds owns application data. A local user gives the project a stable place for the public username, page images, Razorpay configuration, and future campaign ownership. Email is currently the link, though a production model should store the provider name and provider account ID because emails can change.

### Why keep `done: false` first?

Creating an order does not mean money was paid. The pending state separates “checkout started” from “provider verified payment.” Only after a signed server callback/webhook should the record become completed and appear in totals.

### Why sort and limit payments?

`fetchpayments` sorts by amount descending and limits the result to ten. This implements the supporter leaderboard and prevents the initial page from returning an unbounded payment history. A full history would use pagination and a separate aggregate query for total amount/count.

### How are totals calculated?

The client receives up to ten completed payments and uses `reduce` to sum their amounts. This means the displayed total is actually the total of the top ten returned payments, not necessarily all completed contributions. For a correct overall total, MongoDB should use an aggregation pipeline such as `$match` plus `$group`, while the leaderboard remains a separate sorted/limited query.

### What happens when a username changes?

`updateProfile` checks for a collision, updates the user, and rewrites `Payment.to_user` from the old username to the new one. This keeps the contribution list connected to the new page. An immutable `creatorId` foreign key would avoid the bulk rewrite and reduce consistency risk.

### How does the app support many users at once?

Requests do not store creator state in a server process; creator identity comes from the route/session and durable data lives in MongoDB. That makes individual Next.js instances largely stateless and horizontally scalable. MongoDB handles simultaneous document operations, while each Razorpay order has a unique provider order ID. For robust concurrency, the schema still needs unique indexes and idempotent conditional updates.

## Questions an interviewer is likely to ask

### “Why Next.js instead of separate React and Express apps?”

Next.js provided routing, server components, API route handlers, and server actions in one codebase. For this scope, that reduced deployment and integration overhead while still keeping browser code separate from database/payment-secret operations.

### “Why MongoDB?”

The creator and payment documents are straightforward to model with Mongoose, and MongoDB fits an early product whose profile fields may evolve. Payment integrity still benefits from relational-style practices: unique indexes, immutable references, transactions where needed, and explicit state transitions.

### “How do you know a payment succeeded?”

The authoritative proof must be a valid Razorpay signature received by a server endpoint or webhook, followed by provider-state validation. A frontend success message or redirect query parameter is not proof. In the current repository the handler is the main missing payment piece.

### “How do you prevent duplicate payment processing?”

Make the Razorpay order/payment ID unique, then perform a conditional update such as “change created to captured only if it is not already captured.” Webhook retries then return success without applying the payment twice.

### “What makes the dashboard real time?”

In the current version, values are dynamic at page load but not pushed live. True real-time behavior would use polling, SSE, or WebSockets after webhook verification. Say this plainly; do not claim sockets that are not in the code.

### “Where is SSR used?”

The dynamic username route runs on the server for existence validation, and metadata is generated on the server. The interactive contribution UI is a client component and currently fetches its main data after mount. A next improvement is to load safe creator/payment data in the server component.

### “How do you protect Razorpay secrets?”

Order creation uses the secret on the server, which is the right starting point. The current `fetchuser` response accidentally includes the secret, so production hardening requires a public field projection, server-side authorization for credential changes, encrypted storage, and ideally Razorpay's connected-account/OAuth model if applicable.

### “What indexes would you add?”

- unique index on `User.email`;
- unique index on `User.username`;
- unique index on `Payment.oid`;
- compound index on `{ creatorId: 1, status: 1, amount: -1 }`; and
- possibly `{ creatorId: 1, createdAt: -1 }` for payment history.

### “What failure cases did you consider?”

- Razorpay creates an order but the MongoDB insert fails;
- a pending order is never paid;
- callbacks are delayed, duplicated, forged, or received out of order;
- the username changes while a payment is active;
- two users choose the same username concurrently;
- a client changes the amount or recipient in the request; and
- MongoDB or the payment provider is temporarily unavailable.

The remedies are validation, unique constraints, immutable IDs, idempotency, webhooks, reconciliation jobs, retries with bounded backoff, and good audit logs.

## Honest review of the current implementation

An interviewer may inspect the repository, so know these points.

| Area | Good foundation | Current limitation | Production improvement |
|---|---|---|---|
| Authentication | GitHub OAuth through NextAuth | Sign-in lookup uses the wrong callback variable; protected actions lack server auth | Use `user.email`, connect explicitly, call `getServerSession`, authorize ownership |

# Current GetFunds architecture

**Status:** current runtime after the Razorpay Test Mode, creator discovery, and GridFS portfolio upgrade.

GetFunds is a no-real-money portfolio application. It uses Razorpay's official Test Mode Checkout and mock bank screens, but the server explicitly rejects every key that does not begin with `rzp_test_`.

## Razorpay Test Mode payment lifecycle

```mermaid
sequenceDiagram
    autonumber
    actor Visitor
    participant UI as PaymentPage
    participant Order as POST /api/razorpay/order
    participant DB as MongoDB Payment
    participant RP as Razorpay Test API
    participant Checkout as Razorpay Checkout
    participant Verify as POST /api/razorpay/verify

    Visitor->>UI: Enter name, message, and test amount
    UI->>Order: Send untrusted contribution input
    Order->>Order: Validate fields and require rzp_test key
    Order->>RP: Create INR order in paise
    RP-->>Order: order_id
    Order->>DB: Insert pending, done=false record
    Order-->>UI: Return public key + order details
    UI->>Checkout: Open official Test Checkout
    Visitor->>Checkout: Choose mock success or failure
    Checkout-->>UI: payment_id + order_id + signature
    UI->>Verify: Send returned fields
    Verify->>Verify: HMAC-SHA256(server order_id|payment_id)
    Verify->>RP: Fetch payment status and amount
    RP-->>Verify: Test payment details
    Verify->>DB: Atomic done=false to done=true update
    Verify-->>UI: Verified test success
    UI->>UI: Refresh totals, progress, and leaderboard
```

Important security decisions:

- `RAZORPAY_KEY_SECRET` never reaches the browser.
- Every contribution creates an order on the server before Checkout opens.
- The amount sent to Razorpay is derived from validated server input and converted from rupees to paise.
- Verification uses the order ID stored by the server, not an untrusted replacement.
- Signatures use HMAC-SHA256 and `timingSafeEqual`.
- The verification route also checks remote order ID, amount, currency, and payment status.
- Completion is idempotent: only a row with `done:false` can transition to completed.
- Unfinished test orders receive a 24-hour TTL; verified rows remove that expiry.
- Only `done:true` rows contribute to public analytics.
- Live Razorpay keys are rejected intentionally.

Razorpay Test Mode can display card, wallet, netbanking, and UPI mock screens. `success@razorpay` and `failure@razorpay` simulate the two UPI outcomes. No real funds are charged or settled.

## Editable creator funding target

`User.fundingGoal` is a validated number between ₹100 and ₹1,00,00,000. The creator edits it in the protected dashboard. `updateProfile` obtains ownership from the NextAuth session, ignores browser-supplied identity fields, and updates only an explicit allow-list.

The public progress calculation is:

```text
progress percentage = min(100, round(completed test funding / creator funding goal × 100))
```

₹10,000 is only the schema default for a new creator; it is no longer an immutable UI constant.

## Creator search and similar-name suggestions

The global navigation includes a debounced search box:

1. Queries shorter than two characters do not hit MongoDB.
2. Regex metacharacters are escaped before a database query is built.
3. Username/display-name prefix matches are returned first.
4. Remaining slots use substring matches as similar-name recommendations.
5. Only `_id`, display name, username, and profile image are serialized.
6. Selecting a result opens `/{username}`.

Every search result is a normal public creator page. Visitors can view project folders and start a Razorpay test contribution without signing in.

## Project folders and GridFS

A project folder is a `Project` document owned by immutable `creatorId`. It contains title, description, timestamps, and safe file metadata. Actual bytes are stored in the `projectFiles` GridFS bucket.

Why GridFS instead of `public/uploads/{username}`:

- a deployed Next.js filesystem may be read-only or ephemeral;
- username changes should not move files;
- database ownership is easier to enforce;
- large file bytes do not consume a normal MongoDB document's 16 MB limit; and
- files can be streamed instead of loading the entire stored object into the response.

Upload controls:

- creator must have an authenticated server session;
- 1-5 files per folder;
- maximum 5 MB per file;
- maximum 15 MB total per folder;
- allow-listed common code, document, text, archive, and image extensions;
- path separators/control characters are removed from filenames;
- partial GridFS uploads are deleted when an upload fails; and
- project deletion checks ownership and deletes associated GridFS objects.

Public download controls:

- the file ID must exist in a public `Project` document;
- safe images, PDF, Markdown, and plain text may render inline;
- other formats are forced to download;
- `X-Content-Type-Options: nosniff` prevents browser type guessing; and
- HTML/code files cannot execute as GetFunds application pages.

## Honest production limitations

This remains an interview project rather than a live crowdfunding platform. A production system should add persistent distributed rate limiting, abuse protection, malware scanning, object storage/CDN delivery, Razorpay webhooks for browser-drop recovery, captured-payment-only fulfillment, reconciliation jobs, privacy controls, moderation, and a deployment-specific retention policy.

## Interview pitch

> GetFunds is a full-stack crowdfunding portfolio that uses GitHub OAuth for creator identity, server-authorized profile and funding-goal mutations, global creator discovery, and public project portfolios stored with MongoDB GridFS. Visitors can open Razorpay's official Test Checkout on any creator profile. The server creates every order, stores a pending record, verifies the returned HMAC signature with a timing-safe comparison, confirms remote payment fields, and atomically completes the contribution before analytics include it. Live keys are rejected, so the project demonstrates a realistic gateway lifecycle without moving real money.

## Project vocabulary

## Architectural style

The application uses a **full-stack modular monolith**:

- **full-stack**, because React UI and server/database/payment logic live in one Next.js application;
- **modular**, because components, server actions, models, database connection code, and route handlers are separated by responsibility; and
- **monolith**, because these modules are built and deployed as one unit rather than as independent services.

This architecture avoids the operational cost of microservices while the product is small. The main risk is that access-control and data-serialization boundaries can become unclear, so server actions must be treated as public server entry points and secured accordingly.

## Layer-by-layer explanation

### 1. Presentation layer

The App Router supplies pages and layouts. `app/layout.js` wraps every page with `SessionProvider`, `Navbar`, and `Footer`. Components marked `"use client"` handle state, effects, clicks, navigation, toasts, and Razorpay Checkout.

| UI module | Responsibility | Rendering behavior |
|---|---|---|
| `app/page.js` | Landing page | Server component by default |
| `app/about/page.js` | Product information | Server component by default |
| `app/[username]/page.js` | Validate username and provide dynamic metadata | Server component |
| `components/PaymentPage.js` | Contribution form, totals, and leaderboard | Client component |
| `components/Dashboard.js` | Creator profile/settings form | Client component |
| `app/login/page.js` | Starts GitHub OAuth | Client component |
| `components/Navbar.js` | Session-aware navigation | Client component |

### 2. Application/service layer

`actions/useractions.js` is the application boundary for domain operations:

- `fetchuser(username)` loads a creator;
- `fetchpayments(username)` loads up to ten completed payments, ordered by amount;
- `updateProfile(data, oldusername)` updates a creator and migrates payment ownership when the username changes; and
- `initiate(amount, to_username, paymentform)` creates a Razorpay order and a pending local payment.

Because this file starts with `"use server"`, Next.js exposes exported actions to client components through generated RPC-style calls. This keeps SDK secrets and Mongoose code out of the browser bundle, but it does not automatically provide authorization or validation. Each action must still verify the session, validate data, and return only safe fields.

### 3. Authentication layer

`app/api/auth/[...nextauth]/route.js` configures GitHubProvider and exports the NextAuth handler for both GET and POST. The sign-in callback provisions a local user; the session callback adds the local username to `session.user.name`.

The route name is a catch-all dynamic API route, so it handles endpoints such as sign-in, callback, sign-out, session lookup, and CSRF operations under `/api/auth/*`.

### 4. Persistence layer

`db/connectDB.js` opens the Mongoose connection from `MONGO_URI`. Model files use the common hot-reload-safe pattern:

```js
mongoose.models.User || model("User", UserSchema)
```

That prevents model recompilation errors during Next.js development reloads.

### 5. Payment integration layer

Razorpay has two halves:

- the server SDK creates an order with secret credentials; and
- the checkout script collects payment details in Razorpay's UI.

The browser should receive only the public key ID, order ID, amount, and display configuration. The key secret must remain server-side. Payment success must be verified on the server; a browser redirect alone is not trustworthy.

```mermaid

    USER {
```

The relationship is implemented through `Payment.to_user -> User.username`, not an ObjectId reference. This makes page queries readable but creates extra work when a username changes: `updateProfile` must rewrite all associated payments.

A production model would normally:

- make `User.email` and `User.username` unique and indexed;
- store an immutable `creatorId` ObjectId in `Payment`;
- make `Payment.oid` unique for idempotency;
- store money as integer paise;
- replace `done` with a payment-status enum;
- encrypt provider credentials at rest or use a connected-account/token model; and
- optionally introduce `Campaign`, allowing one creator to own multiple campaigns.

An expanded relationship would be:

```text
User 1 --- many Campaigns 1 --- many Payments
```

## Rendering architecture and SSR claim

Next.js uses a mix of server and client rendering in this repository:

- `/`, `/about`, `/dashboard`'s route wrapper, and `/[username]` are server components by default;
- `/[username]` queries MongoDB before returning and can emit a server-generated 404;
- `generateMetadata` creates a dynamic title on the server;
- `PaymentPage`, `Dashboard`, `Login`, and `Navbar` are client components; and
- creator details and payment totals are fetched from the client after mount.

Therefore, it is accurate to say the application **uses Next.js server components and server-generated metadata**, helping routing and page-specific SEO. It is too strong to say the complete funding page is SSR, because the primary profile/payment content is not included in the initial server-rendered HTML. Moving those reads into `app/[username]/page.js` and passing safe data as props would make the claim stronger and improve first-content rendering.

## Environment and secret ownership

| Variable | Used for | Correct visibility |
|---|---|---|
| `MONGO_URI` | MongoDB connection | Server only |
| `GITHUB_ID` | GitHub OAuth app identity | Server configuration |
| `GITHUB_SECRET` | GitHub OAuth secret | Server only |
| `NEXTAUTH_SECRET` | Signs/encrypts NextAuth values | Server only |
| `NEXTAUTH_URL` | OAuth callback base URL | Server configuration |
| `NEXT_PUBLIC_URL` | Razorpay callback URL base | Public by naming convention |

Creator Razorpay secrets are stored in MongoDB rather than environment variables because each creator supplies separate credentials. They must still remain server-only and should be encrypted at rest.

## Current non-functional characteristics

### Scalability

Stateless Next.js instances can scale horizontally if sessions remain JWT-based and MongoDB/Razorpay are external. Database connection reuse, indexes, pagination, rate limiting, and caching would become important under load.

### Consistency

Payment initiation spans two systems: Razorpay order creation and MongoDB insertion. If one succeeds and the other fails, reconciliation is needed. Webhooks and an idempotent status update are the standard recovery mechanism.

### Security

The strongest current boundary is server-side Razorpay order creation. The largest gaps are public serialization of creator secrets, missing callback verification, missing server-action authorization, and overly broad profile updates from raw form data.

### Observability

The repository logs MongoDB connection state but has no structured payment/audit logging, metrics, tracing, or alerting. Production payment flows should attach a request/correlation ID and record state transitions without secrets.

## Current architecture versus production target

| Concern | Current repository | Production target |
|---|---|---|
| Campaign model | One public page per username | Separate Campaign entity with goals/status |
| Payment completion | Callback URL referenced, handler absent | Verified callback plus webhook reconciliation |
| Dashboard | Creator profile configuration | Funding KPIs, history, filters, polling or push updates |
| Public data | Full user object returned | Allow-listed public projection |
| Authorization | Mostly client-side redirect | Session checks inside every protected server action |
| SSR | Route validation and metadata on server | Creator/payment content fetched in server component |
| Data integrity | Mutable username relationship | Immutable IDs, unique indexes, transaction/idempotency |

## 2. Authentication and creator onboarding

```mermaid
sequenceDiagram
    autonumber
    actor Creator
    participant Login as Login client page

    participant GH as GitHub OAuth
    participant DB as MongoDB
    participant Dashboard as Dashboard client
```

### Why both GitHub and MongoDB are needed

GitHub authenticates the person but does not contain GetFunds-specific fields such as the cover image, public username, or Razorpay configuration. The local `User` record connects the external identity (`email`) to application-owned data.

### Session behavior

No NextAuth database adapter or explicit session strategy is configured, so NextAuth v4 uses its default JWT-based session strategy for an OAuth-only configuration. The `session` callback replaces `session.user.name` with the database username; components then use that value to build the public URL and fetch the creator profile.

### Current implementation cautions

- The sign-in callback queries `User.findOne({ email: email })`, but for the GitHub provider it should use `user.email`; the callback's `email` argument is associated with the Email provider flow.
- The session callback performs a database query without explicitly calling `connectDb()`.
- Username creation uses the email prefix, so collisions are possible and should be handled.
- The dashboard redirects inside an effect whose dependency list is empty; it should account for NextAuth's loading state.

## 3. Creator profile update

```mermaid
sequenceDiagram

    participant UI as Dashboard client
```

The UI currently always displays “Profile Updated” and does not inspect the returned `{ error }` value. That should be corrected before presenting username-conflict handling as complete.

The dynamic route itself runs on the server and can return a server-side 404. `generateMetadata` also produces a per-creator title on the server.     Callback-->>Supporter: Redirect to /username?paymentdone=true
```mermaid
    UI->>DB: Fetch completed payments again
    UI-->>Supporter: Show thank-you toast and updated total
```

### Amount handling

Razorpay expects the smallest currency unit. The browser multiplies a rupee value by `100`, and `initiate` sends that paise amount to Razorpay. The stored `Payment.amount` divides it by `100`, so MongoDB stores the display value in rupees.

Example:

```text
supporter enters ₹50 -> browser sends 5000 paise -> Razorpay order is 5000 -> MongoDB stores 50
```

For financial software, storing an integer number of paise is safer than storing rupees because integer arithmetic avoids rounding ambiguity.

### Why create a pending record first?

The pending row connects the locally known donation details to Razorpay's order ID. The callback can later locate that exact row using `oid` and mark it complete only after verification. Reads deliberately use `{ done: true }`, preventing unverified orders from appearing as money raised.

### Production-safe completion rules

A complete callback/webhook should:

1. parse the Razorpay order ID, payment ID, and signature;
2. load the pending payment and the correct creator credentials;
3. verify the HMAC signature with the server-side secret using a timing-safe comparison;
4. confirm the amount, currency, order ID, and captured/authorized state;
5. update by unique order/payment ID using an idempotent operation;
6. never trust a `paymentdone=true` query parameter as proof of payment; and
7. log failures without logging secrets or sensitive payment data.

## 6. Payment state machine

```mermaid
stateDiagram-v2
    [*] --> OrderRequested: supporter clicks Pay
    OrderRequested --> Pending: Razorpay order created and Payment inserted
    OrderRequested --> FailedToStart: order API or database write fails
    Pending --> CancelledOrExpired: supporter cancels or order expires
    Pending --> Verified: valid callback or webhook signature
    Pending --> Rejected: invalid signature or mismatched details
    Verified --> Displayed: fetchpayments includes done=true
    Displayed --> Displayed: duplicate callback is handled idempotently
```

The current schema has only a Boolean `done`, so `CancelledOrExpired`, `Rejected`, and error reasons are conceptual states. A production schema would normally use a status enum such as `created`, `authorized`, `captured`, `failed`, and `refunded`, plus provider payment IDs and timestamps.

## What is inside the system boundary?

GetFunds is a monolithic full-stack Next.js application. The same deployable application contains:

- public pages and React client components;
- server-rendered route logic;
- NextAuth.js authentication endpoints and session callbacks;
- server actions for creator and payment operations;
- Mongoose models and MongoDB access; and
- server-side Razorpay order creation.

This is a useful architecture for a small product because the UI and backend share one repository, deployment, language, and domain model. It is not a pure browser-only app: secrets and database operations run on the server side.

## Actors

### Creator

A creator signs in with GitHub, receives a GetFunds user record, and edits a profile in the dashboard. The profile stores a username, public images, and Razorpay credentials. The username becomes the public route, for example `/alice`.

### Supporter

A supporter does not need to authenticate in the current implementation. They open a creator's public page, see the ten largest completed contributions, enter a name/message/amount, and launch Razorpay Checkout.

## External systems

### GitHub OAuth

GitHub proves the creator's identity. NextAuth.js handles the OAuth endpoints, provider redirects, and session cookie/JWT mechanics. On a successful sign-in, the application tries to find or create a local MongoDB `User` record.

### MongoDB

MongoDB is the system of record for:

- creator identity and page configuration in the `users` collection; and
- contribution/order state in the `payments` collection.

Mongoose supplies schemas and query methods. The application connects using `MONGO_URI`.

### Razorpay

Razorpay creates and processes INR payment orders. The server action uses the target creator's Razorpay key ID and secret to create the order; the browser then opens Razorpay Checkout using the returned order ID.

## Core business capability

Conceptually, the application maps many creators to many fundraising pages:

```text
one User record -> one username -> one public fundraising page -> many Payment records
```

This lets several creators receive contributions independently. It does **not** currently model multiple named campaigns, goals, deadlines, or rewards for one creator. If an interviewer asks about “concurrent campaigns,” explain that the current version supports concurrent creator fundraising pages; a dedicated `Campaign` entity would be the next step for multiple campaigns per creator.

## High-level request paths

| Capability | Entry point | Server-side work | Stored result |
|---|---|---|---|
| GitHub login | `/login` | NextAuth exchanges OAuth data and runs callbacks | A `User` may be created; a session is issued |
| Profile setup | `/dashboard` | `fetchuser` and `updateProfile` server actions | `User` fields are updated |
| Public page | `/{username}` | Server component checks that the user exists; client fetches user/payments | No write |
| Start donation | Pay button | `initiate` creates a Razorpay order and pending `Payment` | `Payment.done` starts as `false` |
| Complete donation | Razorpay callback | Expected: verify signature and update the matching order | Not implemented in this repository |

## Trust boundaries

Three boundaries matter in an interview:

1. **Browser to Next.js server**: browser input is untrusted and must be validated and authorized.
2. **Next.js to MongoDB**: only server code should hold `MONGO_URI` and perform database writes.
3. **Next.js/Razorpay boundary**: order creation and payment verification require server-held secrets and cryptographic signature verification.

The current project correctly creates Razorpay orders on the server, but `fetchuser` also serializes the full user record—including `razorpaysecret`—to public client code. This must be removed in a production version by returning a safe public projection and keeping secrets server-only.

let razorpayLoaderPromise;

const loadRazorpayCheckout = () => {
  if (typeof window === "undefined") {
    return Promise.reject(new Error("Razorpay Checkout requires a browser."));
  }

  if (window.Razorpay) {
    return Promise.resolve();
  }

  if (razorpayLoaderPromise) {
    return razorpayLoaderPromise;
  }

  razorpayLoaderPromise = new Promise((resolve, reject) => {
    let script = document.getElementById(RAZORPAY_SCRIPT_ID);
    const shouldAppendScript = !script;
    let timeout;

    const cleanup = () => {
      window.clearTimeout(timeout);
      script?.removeEventListener("load", handleLoad);
      script?.removeEventListener("error", handleError);
    };

    const handleLoad = () => {
      cleanup();

      if (window.Razorpay) {
        resolve();
        return;
      }

      razorpayLoaderPromise = undefined;
      reject(new Error("Razorpay loaded without exposing Checkout. Please retry."));
    };

    const handleError = () => {
      cleanup();
      script?.remove();
      razorpayLoaderPromise = undefined;
      reject(
        new Error(
          "Razorpay Checkout was blocked. Check your internet connection or ad blocker, then retry.",
        ),
      );
    };

    if (!script) {
      script = document.createElement("script");
      script.id = RAZORPAY_SCRIPT_ID;
      script.src = "https://checkout.razorpay.com/v1/checkout.js";
      script.async = true;
    }

    script.addEventListener("load", handleLoad, { once: true });
    script.addEventListener("error", handleError, { once: true });

    timeout = window.setTimeout(() => {
      cleanup();
      script?.remove();
      razorpayLoaderPromise = undefined;
      reject(
        new Error(
          "Razorpay Checkout timed out. Check your connection or disable blocking extensions, then retry.",
        ),
      );
    }, 15000);

    if (window.Razorpay) {
      handleLoad();
      return;
    }

    if (shouldAppendScript) {
      document.head.appendChild(script);
    }
  });

  return razorpayLoaderPromise;
};

    $or: [
      { username: containsPattern },
      { name: containsPattern },
    ],
  })
    .select(projection)
    .sort({ username: 1 })
    .limit(8 - prefixMatches.length)
    .lean();

  return [...prefixMatches, ...similarMatches].map(

  );

export const validateProjectFiles = (files) => {
  if (files.length < 1 || files.length > MAX_PROJECT_FILES) {
    return `Choose between 1 and ${MAX_PROJECT_FILES} files.`;
  }

  let totalSize = 0;

  for (const file of files) {
    const extension = file.name.split(".").pop()?.toLowerCase() || "";

    if (!ALLOWED_FILE_TYPES.has(file.type) && !ALLOWED_EXTENSIONS.has(extension)) {
      return `${file.name} uses an unsupported file type.`;
    }

    if (file.size < 1 || file.size > MAX_PROJECT_FILE_BYTES) {
      return `${file.name} must be larger than 0 bytes and no larger than 5 MB.`;
    }

    totalSize += file.size;
  }

    return "A project folder can contain at most 15 MB in total.";
  }
